function saveMemorySilently(text) { let lower = text.toLowerCase(); let itemName = "Gegenstand"; let locationName = "am festgelegten Ort"; // Erweiterte Liste für flexiblere Gegenstände const commonItems = [ 'taschentuch', 'taschentücher', 'tücher', 'handtuch', 'handtücher', 'schlüssel', 'lebenslauf', 'unterlagen', 'akten', 'brille', 'geldbeutel', 'portemonnaie', 'handy', 'ausweis', 'tasche', 'schmuck', 'uhr', 'stift', 'dokumente', 'papiere', 'fernbedienung' ]; for (let item of commonItems) { if (lower.includes(item)) { // Singular/Plural schön formen itemName = item.charAt(0).toUpperCase() + item.slice(1); break; } } if (itemName === "Gegenstand") { let match = lower.match(/(?:meinen|meine|mein|ein|eine|das|den)\s+([a-zäöü]+)/i); if (match && match[1] && !['in', 'auf', 'unter', 'packe', 'lege', 'tue'].includes(match[1])) { itemName = match[1].charAt(0).toUpperCase() + match[1].slice(1); } } let locMatch = text.match(/(?:in\s+den|in\s+die|in\s+dem|im|auf\s+dem|auf\s+der|unter\s+dem|unter\s+der|in|auf|unter)\s+(.*)/i); if (locMatch && locMatch[1]) { let rawLoc = locMatch[1].trim(); locationName = rawLoc; } let nowTime = new Date().toLocaleDateString('de-DE', { weekday: 'short', hour: '2-digit', minute: '2-digit' }); // Wir speichern jetzt den Ort inklusive Zeitstempel, falls es öfters wechselt! memory[itemName.toLowerCase()] = { item: itemName, location: locationName, time: nowTime }; localStorage.setItem('omas_memory', JSON.stringify(memory)); document.getElementById('status').innerText = `🧠 Für ${itemName} gemerkt.`; speakTextOut(`Habe ich mir für ${itemName} gemerkt.`); } function searchMemory(text) { let lower = text.toLowerCase(); let foundKey = null; for (let key in memory) { if (lower.includes(key)) { foundKey = key; break; } } if (foundKey) { let entry = memory[foundKey]; let formattedLoc = entry.location; if (formattedLoc.startsWith('dem ')) { formattedLoc = "im " + formattedLoc.substring(4); } else if (!formattedLoc.startsWith('im ') && !formattedLoc.startsWith('in ') && !formattedLoc.startsWith('auf ')) { formattedLoc = "in " + formattedLoc; } // Antwort mit Silvia und dem letzten Stand inklusive Zeitpunkt let answerText = `Silvia, deine ${entry.item} sind ${formattedLoc}. Das hattest du zuletzt am ${entry.time} Uhr so hingelegt.`; document.getElementById('status').innerText = `💡 ${answerText}`; speakTextOut(answerText); } else { let notFound = `Silvia, dazu habe ich mir nichts im Gedächtnis aufgeschrieben.`; document.getElementById('status').innerText = notFound; speakTextOut(notFound); } }
